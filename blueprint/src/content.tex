\chapter{Foundations}

\begin{definition}[Second differential]
  \label{def:sndFDeriv}
  \lean{MorseFloer.sndFDeriv}
  \leanok
  For $f$ on a normed space, $(d^2f)_x$ is the derivative of $x \mapsto (df)_x$,
  read as a continuous bilinear map.
\end{definition}

\begin{theorem}[Transformation rule for second differentials]
  \label{thm:sndFDeriv_comp}
  \lean{MorseFloer.sndFDeriv_comp}
  \leanok
  \uses{def:sndFDeriv}
  Under a change of coordinates $\varphi$,
  \[ d^2(f\circ\varphi)_x(h,k) = d^2f_{\varphi(x)}(d\varphi\, h, d\varphi\, k)
     + df_{\varphi(x)}\bigl(d^2\varphi_x(h,k)\bigr). \]
  The correction term lies in the image of $df$, which is exactly why the second
  differential is not chart-independent in general.
\end{theorem}

\begin{corollary}[Tensoriality at a critical point]
  \label{cor:sndFDeriv_critical}
  \lean{MorseFloer.sndFDeriv_comp_of_critical}
  \leanok
  \uses{thm:sndFDeriv_comp}
  Where $df$ vanishes the correction term disappears, so $d^2f$ transforms as a
  bilinear form.  This is what makes the Hessian of a function on a manifold well
  defined at its critical points, and nowhere else.
\end{corollary}

\begin{definition}[Hessian, critical point, Morse function]
  \label{def:hessian}
  \lean{MorseFloer.hessian, MorseFloer.IsCriticalPoint, MorseFloer.IsNondegenerate,
        MorseFloer.IsMorseFunction}
  \leanok
  \uses{cor:sndFDeriv_critical}
  On a manifold, $x$ is critical for $f$ when $d f_x = 0$; the Hessian is the
  second differential of $f$ read in the preferred extended chart at $x$; the
  critical point is nondegenerate when that bilinear form is; and $f$ is a Morse
  function when all its critical points are nondegenerate.
\end{definition}

\begin{definition}[Index of a critical point]
  \label{def:index}
  \lean{MorseFloer.index, MorseFloer.morseIndex}
  \leanok
  \uses{def:hessian}
  The index is the largest dimension of a subspace on which the Hessian quadratic
  form is negative definite, i.e.\ \texttt{sigNeg} of that form.
\end{definition}

\begin{proposition}[The index is well defined]
  \label{prop:index_invariant}
  \lean{MorseFloer.index_eq_of_equivalent}
  \leanok
  \uses{def:index}
  Equivalent quadratic forms have the same index.  This is Sylvester's law of
  inertia, already available in Mathlib, and it is what makes the index
  independent of the coordinates used to compute it.
\end{proposition}

\begin{proposition}[Symmetry of the Hessian]
  \label{prop:hessian_symm}
  \lean{MorseFloer.Chapter1.hessian_symm}
  \leanok
  \uses{def:hessian}
  The Hessian is a symmetric bilinear form.  The book derives this from
  $X\cdot(Y\cdot f) - Y\cdot(X\cdot f) = [X,Y]\cdot f = df([X,Y]) = 0$; in a chart

\section{\S1.2 Existence and multitude of Morse functions}

\begin{definition}[Squared distance to a point]
  \label{def:distSq}
  \lean{MorseFloer.Chapter1.distSq}
  \leanok
  The function $f_p(x) = \lVert x - p\rVert^2$ of \S1.2.a.
\end{definition}

\begin{proposition}[Differential of the squared distance]
  \label{prop:fderiv_distSq}
  \lean{MorseFloer.Chapter1.hasFDerivAt_distSq, MorseFloer.Chapter1.fderiv_distSq,
        MorseFloer.Chapter1.isCriticalPt_distSq_iff}
  \leanok
  \uses{def:distSq}
  $(df_p)_x(\xi) = 2\langle x-p, \xi\rangle$.  On a submanifold $V$ this says $x$
  is critical for $f_p$ exactly when $x-p \perp T_xV$, the characterisation the
  book feeds to Sard's theorem through the normal bundle.
\end{proposition}

\begin{proposition}[Proposition 1.2.1]
  \label{prop:1.2.1}
  \lean{MorseFloer.Chapter1.ae_isMorseFunction_distSq}
  \notready
  \uses{prop:fderiv_distSq}
  For almost every $p \in \mathbb{R}^n$, the function $f_p$ restricted to a
  submanifold $V \subseteq \mathbb{R}^n$ is a Morse function.
  \emph{Assumed:} the proof applies Sard's theorem to the endpoint map of the
  normal bundle, and Sard's theorem for manifolds is not in Mathlib.
\end{proposition}

\begin{lemma}[Lemma 1.2.2]
  \label{lem:1.2.2}
  \notready
  The normal bundle $N = \{(x,v) \in V\times\mathbb{R}^n : v \perp T_xV\}$ is a
  submanifold of $V\times\mathbb{R}^n$, and $p = x+v$ is a critical value of the
  endpoint map $E(x,v) = x+v$ exactly when the second-derivative matrix of $f_p$
  fails to be invertible.
  \emph{Not statable:} Mathlib has no normal bundle of a submanifold and no
  general submanifold type to carry it.
\end{lemma}

\begin{proposition}[Proposition 1.2.4]
  \label{prop:1.2.4}
  \notready

\end{proposition}

\section{\S1.3 The Morse lemma, index of a critical point}

\begin{theorem}[Theorem 1.3.1, the Morse lemma]
  \label{thm:1.3.1}
  \lean{MorseFloer.Chapter1.morse_lemma}
  \notready
  \uses{def:hessian}
  Near a nondegenerate critical point $c$ there are coordinates in which $f$
  equals its quadratic model: a chart $\varphi$ centred at $c$ with
  $f(y) = f(c) + \tfrac12 d^2f_c(\varphi y, \varphi y)$.  Diagonalising by
  Sylvester's law gives the book's normal form
  $f(c) - \sum_{j\le i} x_j^2 + \sum_{j>i} x_j^2$.
  \emph{Assumed:} the book's proof is an induction on dimension via the implicit
  function theorem; not yet carried out in Lean.
\end{theorem}

\begin{corollary}[Corollary 1.3.2]
  \label{cor:1.3.2}
  \lean{MorseFloer.Chapter1.isolated_of_nondegenerate}
  \leanok
  \uses{def:hessian}
  Nondegenerate critical points are isolated.  Proved as Exercise 2 asks,
  \emph{without} the Morse lemma: $df$ has an invertible strict derivative at
  $c$, so the inverse function theorem makes it a local homeomorphism and $c$ is
  its only nearby zero.
\end{corollary}

\begin{proposition}[Remark 1.3.3]
  \label{prop:1.3.3}
  \lean{MorseFloer.Chapter1.index_neg_add_index}
  \leanok
  \uses{def:index}
  A critical point of index $i$ for $f$ has index $n-i$ for $-f$.
\end{proposition}

\begin{proposition}[The torus function, \S1.4.c]
  \label{prop:1.4.c}
  \lean{MorseFloer.Chapter1.cosFactor,
        MorseFloer.Chapter1.deriv_cosFactor,

\section{\S3.1 The complex}

\begin{definition}[The chain groups and the differential]
  \label{def:3.1}
  \lean{MorseFloer.Chapter3.CritSet, MorseFloer.Chapter3.Chains,
        MorseFloer.Chapter3.dLin}
  \leanok
  \uses{def:index}
  $C_k(f)$ is the free module on the critical points of index $k$, and
  $\partial_X a = \sum_b n_X(a,b)\, b$ where $n_X(a,b)$ counts, modulo $2$, the
  trajectories of $X$ from $a$ down to $b$.  The count enters as abstract data
  `cnt`, since the geometry that produces it is not available.
\end{definition}

\begin{definition}[The broken-trajectory hypothesis]
  \label{def:brokenPairs}
  \lean{MorseFloer.Chapter3.BrokenPairs}
  \leanok
  \uses{def:3.1}
  For $\operatorname{Ind} a = \operatorname{Ind} b + 2$, the once-broken
  trajectories from $a$ to $b$ cancel in pairs:
  $\sum_c n_X(a,c)\,n_X(c,b) = 0$.  This is the \emph{conclusion} of
  Theorem 3.2.6 together with the fact that a compact $1$-manifold has an even
  number of boundary points, taken as a hypothesis because its proof is
  geometric.
\end{definition}

  \uses{def:16.2.a}
  In finite dimensions $\operatorname{ind} u = \dim E - \dim F$ whatever $u$ is.
  This is rank--nullity plus the dimension of a quotient, and it is the check
  that the definition is the right one: the index does not see the operator,
  which is why it is a deformation invariant.
\end{proposition}

\begin{theorem}[Theorems 16.2.9, 16.2.10 and Proposition 16.2.7]
  \label{thm:16.2.stability}
  \lean{MorseFloer.Chapter16.fredholmIndex_locally_constant,
        MorseFloer.Chapter16.fredholmIndex_comp,
        MorseFloer.Chapter16.fredholmIndex_add_compact}
  \notready
  \uses{def:16.2.a}
  The index is locally constant, additive under composition, and unchanged by a
  compact perturbation.  \emph{Assumed:} standard, but their proofs need
  operator theory Mathlib does not yet carry.  Chapter 8 uses the compact-
  perturbation statement to reduce the linearised Floer operator to a
  constant-coefficient model whose index is computable.
\end{theorem}

\begin{theorem}[\S16.3--16.4: distributions and Sobolev spaces]
  \label{thm:16.4}
  \notready
  Weak derivatives and convolution of distributions; the spaces $W^{k,p}(U)$
  with their extension, trace, Poincar\'e (16.4.3), Sobolev embedding (16.4.4)
  and Rellich compactness (16.4.6) theorems.
  \emph{Not statable:} Mathlib has neither distributions in this sense nor
  Sobolev spaces on a domain.  This is the gap that puts the elliptic regularity
  of Chapters 12 and 13 out of reach.
\end{theorem}

\chapter{Symplectic geometry (Chapter 5)}

Mathlib has the linear symplectic group but no symplectic manifolds and no
differential forms on manifolds, so the linear theory of \S5.1 and \S5.6 is
formalized in depth while the manifold statements are gaps.

\begin{definition}[Symplectic vector spaces and their subspaces, \S5.1]
  \label{def:5.1}
  \lean{MorseFloer.Chapter5.IsSymplecticForm, MorseFloer.Chapter5.IsIsotropic,
        MorseFloer.Chapter5.IsCoisotropic, MorseFloer.Chapter5.IsLagrangian,
        MorseFloer.Chapter5.IsSymplecticSubspace, MorseFloer.Chapter5.IsSymplecticBasis}
  \leanok
  A nondegenerate alternating form, and the isotropic, coisotropic, Lagrangian
  and symplectic subspaces it distinguishes.
\end{definition}

\begin{proposition}[Proposition 5.1.1, existence of a symplectic basis]
  \label{prop:5.1.1}
  \lean{MorseFloer.Chapter5.exists_isSymplecticBasis,
        MorseFloer.Chapter5.even_finrank_of_isSymplecticForm}
  \leanok
  \uses{def:5.1}
  Every symplectic vector space has a symplectic basis, so its dimension is
  even.  Proved in full by the book's induction: pick $e,f$ with
  $\omega(e,f)=1$, note $\omega$ restricts nondegenerately to
  $\langle e,f\rangle^\perp$, and recurse.
\end{proposition}

\begin{proposition}[Dimension counts for subspaces, \S5.1]
  \label{prop:5.1.dim}
  \lean{MorseFloer.Chapter5.finrank_orthogonal_add,
        MorseFloer.Chapter5.orthogonal_orthogonal,
        MorseFloer.Chapter5.IsLagrangian.two_mul_finrank,
        MorseFloer.Chapter5.isCoisotropic_iff_isotropic_orthogonal,
        MorseFloer.Chapter5.isSymplecticSubspace_iff_isCompl}
  \leanok
  \uses{prop:5.1.1}
  The symplectic orthogonal is an involution of the right dimension; a
  Lagrangian is exactly half-dimensional; a subspace is symplectic exactly when
  it is complementary to its orthogonal.
\end{proposition}

\begin{definition}[Hamiltonian vector fields, \S5.4]
  \label{def:5.4}
  \lean{MorseFloer.Chapter5.hamiltonianVector, MorseFloer.Chapter5.hamiltonianField,
        MorseFloer.Chapter5.hamiltonianVector_spec,
        MorseFloer.Chapter5.hamiltonianVector_eq_zero_iff,
        MorseFloer.Chapter5.apply_hamiltonianVector,
        MorseFloer.Chapter5.hamiltonianVector_stdForm}
  \leanok
  \uses{def:5.1}
  $X_H$ is defined by $\omega(Y,X_H) = dH(Y)$ and is unique; its zeros are the
  critical points of $H$; $H$ is conserved along it; and on $\mathbb{R}^{2n}$
  the definition unwinds to Hamilton's equations $X_H = J\,\nabla H$.
\end{definition}

\begin{theorem}[Theorem 5.3.2, Darboux]
  \label{thm:5.3.2}
  \lean{MorseFloer.Chapter5.darboux}
  \notready
  \uses{def:5.1}
  Every symplectic form is locally the standard one.  \emph{Assumed}, stated in
  the local model; the manifold statement needs differential forms.
\end{theorem}

\begin{proposition}[\S5.5, calibrated complex structures]
  \label{prop:5.5}
  \lean{MorseFloer.Chapter5.IsCalibrated, MorseFloer.Chapter5.calibratedMetric,
        MorseFloer.Chapter5.calibratedMetric_pos, MorseFloer.Chapter5.stdJ_isCalibrated}
  \leanok
  \uses{def:5.1}
  A complex structure compatible with $\omega$ induces an inner product
  $\omega(\cdot,J\cdot)$; the standard $J_0$ on $\mathbb{R}^{2n}=\mathbb{C}^n$ is
  calibrated and induces the Euclidean product.  Existence for a general
  $\omega$ (Lemma 5.5.3) and contractibility of the space of such $J$
  (Proposition 5.5.4) are assumed: the first needs polar decomposition, which
  Mathlib lacks.
\end{proposition}

\begin{proposition}[\S5.6, the symplectic group]
  \label{prop:5.6}
  \lean{MorseFloer.Chapter5.mem_symplecticGroup_iff_preserves,
        MorseFloer.Chapter5.symplectic_of_orthogonal_of_complexLinear,
        MorseFloer.Chapter5.det_eq_one_of_symplectic,
        MorseFloer.Chapter5.transpose_mul_J_eq,
        MorseFloer.Chapter5.exp_mem_symplecticGroup}
  \leanok
  \uses{prop:5.5}
  Mathlib's \texttt{Matrix.symplecticGroup} is the isometry group of the standard
  form; symplectic, orthogonal and complex-linear --- any two imply the third
  (Proposition 5.6.2); symplectic matrices have determinant one (Corollary
  5.6.10); and the flow of a quadratic Hamiltonian is symplectic.  That
  $Sp(2n)$ retracts onto $U(n)$ with $\pi_1 \simeq \mathbb{Z}$ (Proposition
  5.6.9) --- the input to the Maslov index of Chapter 7 --- is a gap: $U(n)$ is
  not available as a subgroup of $GL(2n;\mathbb{R})$.
\end{proposition}
