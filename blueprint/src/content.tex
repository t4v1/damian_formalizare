\chapter{Foundations}

\begin{definition}[Second differential]
  \label{def:sndFDeriv}
  \lean{MorseFloer.sndFDeriv}
  \leanok
  For $f$ on a normed space, $(d^2f)_x$ is the derivative of $x \mapsto (df)_x$,
  read as a continuous bilinear map.
\end{definition}

\begin{theorem}[Transformation rule for second differentials]
  \label{thm:sndFDeriv_comp}
  \lean{MorseFloer.sndFDeriv_comp}
  \leanok
  \uses{def:sndFDeriv}
  Under a change of coordinates $\varphi$,
  \[ d^2(f\circ\varphi)_x(h,k) = d^2f_{\varphi(x)}(d\varphi\, h, d\varphi\, k)
     + df_{\varphi(x)}\bigl(d^2\varphi_x(h,k)\bigr). \]
  The correction term lies in the image of $df$, which is exactly why the second
  differential is not chart-independent in general.
\end{theorem}

\begin{corollary}[Tensoriality at a critical point]
  \label{cor:sndFDeriv_critical}
  \lean{MorseFloer.sndFDeriv_comp_of_critical}
  \leanok
  \uses{thm:sndFDeriv_comp}
  Where $df$ vanishes the correction term disappears, so $d^2f$ transforms as a
  bilinear form.  This is what makes the Hessian of a function on a manifold well
  defined at its critical points, and nowhere else.
\end{corollary}

\begin{definition}[Hessian, critical point, Morse function]
  \label{def:hessian}
  \lean{MorseFloer.hessian, MorseFloer.IsCriticalPoint, MorseFloer.IsNondegenerate,
        MorseFloer.IsMorseFunction}
  \leanok
  \uses{cor:sndFDeriv_critical}
  On a manifold, $x$ is critical for $f$ when $d f_x = 0$; the Hessian is the
  second differential of $f$ read in the preferred extended chart at $x$; the
  critical point is nondegenerate when that bilinear form is; and $f$ is a Morse
  function when all its critical points are nondegenerate.
\end{definition}

\begin{definition}[Index of a critical point]
  \label{def:index}
  \lean{MorseFloer.index, MorseFloer.morseIndex}
  \leanok
  \uses{def:hessian}
  The index is the largest dimension of a subspace on which the Hessian quadratic
  form is negative definite, i.e.\ \texttt{sigNeg} of that form.
\end{definition}

\begin{proposition}[The index is well defined]
  \label{prop:index_invariant}
  \lean{MorseFloer.index_eq_of_equivalent}
  \leanok
  \uses{def:index}
  Equivalent quadratic forms have the same index.  This is Sylvester's law of
  inertia, already available in Mathlib, and it is what makes the index
  independent of the coordinates used to compute it.
\end{proposition}

  \label{prop:hessian_symm}
  \lean{MorseFloer.Chapter1.hessian_symm}

\begin{definition}[Squared distance to a point]
  \label{def:distSq}
  \lean{MorseFloer.Chapter1.distSq}
  \leanok
  The function $f_p(x) = \lVert x - p\rVert^2$ of \S1.2.a.
\end{definition}

\begin{proposition}[Differential of the squared distance]
  \label{prop:fderiv_distSq}
  \lean{MorseFloer.Chapter1.hasFDerivAt_distSq, MorseFloer.Chapter1.fderiv_distSq,
        MorseFloer.Chapter1.isCriticalPt_distSq_iff}
  \leanok
  \uses{def:distSq}
  $(df_p)_x(\xi) = 2\langle x-p, \xi\rangle$.  On a submanifold $V$ this says $x$
  is critical for $f_p$ exactly when $x-p \perp T_xV$, the characterisation the
  book feeds to Sard's theorem through the normal bundle.
\end{proposition}

\begin{proposition}[Proposition 1.2.1]
  \label{prop:1.2.1}
  \lean{MorseFloer.Chapter1.ae_isMorseFunction_distSq,
        MorseFloer.ae_isMorseFunction_distSq_general}
  \leanok
  \uses{prop:fderiv_distSq, thm:14.2.1}
  For almost every $p \in \mathbb{R}^n$, the function $f_p$ restricted to a
  submanifold $V \subseteq \mathbb{R}^n$ is a Morse function.
  \emph{Proved} (\texttt{MorseFloer/Part1/DistSqMorse.lean}), without the normal
  bundle as a global object.  $V$ is second countable, being embedded in
  $\mathbb{R}^n$, so countably many small chart domains cover it.  On each, a
  smoothly varying projection onto the normal spaces turns the endpoint map
  $(u,w) \mapsto g(u) + P(u)w$ into a map between spaces of dimension $n$, and
  Mathlib's \emph{equidimensional} Sard (\ref{thm:14.2.1}) says its critical
  values are null.  The local half of Lemma \ref{lem:1.2.2} is proved along the
  way: a kernel vector of the Hessian of $f_p$ at a critical point yields one of
  the differential of the endpoint map.
\end{proposition}

\section{\S1.3 The Morse lemma, index of a critical point}

\begin{theorem}[Theorem 1.3.1, the Morse lemma]
  \label{thm:1.3.1}
  \lean{MorseFloer.Chapter1.morse_lemma,
        MorseFloer.Chapter1.morse_lemma_real,
        MorseFloer.Chapter1.morse_lemma_dim_one,
        MorseFloer.Chapter1.exists_analyticAt_sq_factor,
        MorseFloer.morse_lemma_general}
  \leanok
  \uses{def:hessian}
  Near a nondegenerate critical point $c$ there are coordinates in which $f$
  equals its quadratic model: a chart $\varphi$ centred at $c$ with
  $f(y) = f(c) + \tfrac12 d^2f_c(\varphi y, \varphi y)$.  Diagonalising by
  Sylvester's law gives the book's normal form
  $f(c) - \sum_{j\le i} x_j^2 + \sum_{j>i} x_j^2$.
  \emph{Proved in every finite dimension}
  (\texttt{MorseFloer/Part1/MorseLemma.lean}), by the analyst's route rather
  than the book's induction.  Hadamard's lemma comes from the Taylor formula
  with integral remainder along segments, $Q_x = \int_0^1 (1-t)\,
  d^2f(c+tx)\,dt$; the operator $(Q_0)^{-1}Q_x$, which is near the identity, has
  a square root $A_x$ obtained from the inverse function theorem inside the
  $Q_0$-self-adjoint operators, Lipschitz in $x$; and $y \mapsto A_{y-c}(y-c)$
  has strict derivative the identity at $c$, so it is a chart.  Since the
  statement asks only for a homeomorphism, no smooth dependence on the
  parameter is needed.  The one-dimensional case \texttt{morse\_lemma\_real},
  proved earlier through the analytic factorisation
  $f(y) = f(c) + (y-c)^2 g(y)$, is kept.
\end{theorem}

\begin{corollary}[Corollary 1.3.2]
  \label{cor:1.3.2}
  \lean{MorseFloer.Chapter1.isolated_of_nondegenerate}
  \leanok
  \uses{def:hessian}
  Nondegenerate critical points are isolated.  Proved as Exercise 2 asks,
  \emph{without} the Morse lemma: $df$ has an invertible strict derivative at
  $c$, so the inverse function theorem makes it a local homeomorphism and $c$ is

\end{corollary}

\begin{theorem}[Theorem 14.2.1, Sard]
  \label{thm:14.2.1}
  \lean{MorseFloer.Chapter14.criticalSet, MorseFloer.Chapter14.criticalValues,
        MorseFloer.Chapter14.not_surjective_of_mem_criticalSet,
        MorseFloer.Chapter14.det_eq_zero_iff_not_surjective,
        MorseFloer.Chapter14.sard_of_finrank_lt,
        MorseFloer.Chapter14.sard_of_finrank_eq,
        MorseFloer.Chapter14.sard_of_lt_finrank,
        MorseFloer.Chapter14.sard}
  \leanok
  \uses{def:14.2.a}
  The critical values of a smooth map form a set of measure zero.

  \emph{Proved.}  The proof splits on the two dimensions.  When
  $\dim E < \dim F$ the entire image is null and criticality is irrelevant: a
  $C^1$ map does not raise Hausdorff dimension, and a set of Hausdorff dimension
  below $\dim F$ is null.  When $\dim E = \dim F$ criticality means a vanishing
  determinant, and the statement is Mathlib's Jacobian lemma after transport
  along a linear isomorphism; differentiability alone suffices here, without the
  book's $C^1$.

  When $\dim E > \dim F$ the statement is the genuine Morse--Sard theorem,
  stated at the sharp smoothness threshold $k \ge \dim E - \dim F + 1$, which
  Whitney's 1935 example shows cannot be lowered.  It is derived from Moreira's
  sharper Hausdorff-measure version, transplanted from Kudryashov's
  \texttt{SardMoreira} project into \texttt{Part2/SardMoreira/}: with rank bound
  $\dim F - 1$ and H\"older exponent $0$ the critical values are null for
  $\mathcal{H}^{\dim F}$, which is a Haar measure on $F$, so they are null for
  every Haar measure.
\end{theorem}

\begin{proposition}[Proposition 2.1.6]
  \label{prop:2.1.6}
  \lean{MorseFloer.Chapter2.exists_isCriticalPt_tendsto,

\begin{corollary}[Corollary 2.1.9, Reeb's theorem]
  \label{cor:2.1.9}
  \lean{MorseFloer.Chapter2.reeb, MorseFloer.reeb_general,
        MorseFloer.exists_flow,
        MorseFloer.Reeb.exists_gradientLike_field,
        MorseFloer.Reeb.exists_level_sphere,
        MorseFloer.Reeb.GradientLikeFlow.nonempty_homeomorph_sphere}
  \leanok
  \uses{def:hessian, thm:1.3.1}
  A compact manifold carrying a Morse function with exactly two critical points
  is homeomorphic to a sphere.  Stated for a smooth $f$ on a compact
  \emph{Hausdorff} manifold: a chart atlas does not imply Hausdorff, and the
  circle with a doubled point is a counterexample without it.
  \emph{Proved} (\texttt{MorseFloer/Part1/Reeb.lean}), but not by the book's
  gluing of two disks along \ref{thm:2.1.7}, which would need diffeomorphisms
  between sublevel sets.  A partition of unity glues coordinate gradients into a
  $C^1$ field $v$ with $df(v) > 0$ off the two critical points, the minimum $m$
  and the maximum $M$.  Its flow (\texttt{MorseFloer/Part1/ManifoldFlow.lean}:
  Picard--Lindel\"of in charts, with continuous dependence on the initial point,
  made global by compactness) carries every other point, continuously, to one
  level $f = c$ just above $f(m)$.  The Morse lemma at $m$ identifies that level
  with $S^{n-1}$, and together with the height of the point this gives a
  continuous bijection $V \to S^n$, which is a homeomorphism since $V$ is
  compact.
\end{corollary}

\section{\S2.3 Dimension one}

\begin{theorem}[Theorem 2.3.2]
  \label{thm:2.3.2}
  \lean{MorseFloer.Chapter2.classification_dim_one,
        MorseFloer.OneManifold.classification_dim_one_general,
        MorseFloer.OneManifold.nonempty_homeomorph_sphere_of_chartedSpace}
  \leanok
  A compact connected $1$-manifold is a circle or a segment --- so a compact
  $1$-manifold has an even number of boundary points, which is exactly what
  \ref{def:brokenPairs} needs.
  \emph{Proved for the closed case} (\texttt{MorseFloer/Part1/OneManifold.lean}),
  up to homeomorphism, for any compact connected Hausdorff space locally
  homeomorphic to $\mathbb{R}$; the Hausdorff hypothesis is necessary, as the
  circle with a doubled point shows.  The proof is Gale's topological one, not
  the book's Morse-theoretic one.  Where two open arcs overlap without either
  containing the other, the overlap is at most two rays, so their union is an
  arc or a circle; merging a finite cover by arcs ends in the circle.  The case
  with boundary is not stated, since it needs the manifolds with boundary of
  \S2.3.a.
\end{theorem}

\begin{theorem}[Theorem 2.3.3, Brouwer]
  \label{thm:2.3.3}
  \lean{MorseFloer.Chapter2.brouwer, MorseFloer.brouwer_fixed_point,
        MorseFloer.no_retraction, MorseFloer.Brouwer.no_C1_retraction,
        MorseFloer.Chapter2.brouwer_dim_zero, MorseFloer.Chapter2.brouwer_dim_one}
  \leanok
  A continuous self-map of the closed unit ball has a fixed point.
  \emph{Proved in every dimension} (\texttt{MorseFloer/Part1/Brouwer.lean}), but
  not by the book's route, which runs through Sard and \ref{thm:2.3.2}, nor by
  any homological one: this Mathlib has no homology of spheres.  The proof is
  analytic, after Milnor and Rogers.  A $C^1$ retraction $r$ of the ball $D$
  onto its boundary would give $\int_D \det Dr = \operatorname{vol} D$: follow
  the homotopy $\mathrm{id} + t(r - \mathrm{id})$, a bijection of $D$ for small
  $t$, with the change of variables formula, and note that the integral is a
  polynomial in $t$.  It would also give $\int_D \det Dr = 0$, because
  $\|r\| = 1$ makes $Dr$ singular.  Smooth partitions of unity reduce a
  continuous fixed-point-free map to a $C^1$ one, and the ray construction turns
  that into a $C^1$ retraction.
\end{theorem}

\begin{proposition}[Proposition 16.2.7, compact perturbations]
  \label{prop:16.2.7}
  \lean{MorseFloer.Chapter16.fredholmIndex_add_compact,
        MorseFloer.Chapter16.isFredholm_add_and_fredholmIndex_eq_of_hasNoetherianRange,

  $q(u+k) = 1 + (qu-1) + qk$.  The general case is Riesz--Schauder, that $1+c$

  alternative by a counting argument that needs neither Riesz's ascent--descent

\begin{theorem}[Theorem 5.3.2, Darboux]
  \label{thm:5.3.2}
  \lean{MorseFloer.Chapter5.darboux, MorseFloer.Darboux.darboux,
        MorseFloer.Darboux.poincare_d, MorseFloer.Darboux.exists_vectorField,
        MorseFloer.Darboux.exists_pullback, MorseFloer.FlowC1.exists_flow,
        MorseFloer.FlowC1.exists_flow_of_lipschitz,
        MorseFloer.FlowC1.exists_fderiv_flow}
  \leanok
  \uses{def:5.1}
  Every symplectic form is locally the standard one.  Proved in the local model
  for a \emph{smooth} closed form (without smoothness the closedness condition is
  vacuous where $\omega$ is not differentiable, and the statement is false); the
  manifold statement needs differential forms.  The proof is Moser's path method.
  The Poincar\'e lemma gives $\omega - \omega_{x_0} = d\alpha$ with
  $\alpha_x = \int_0^1 s\,(\omega - \omega_{x_0})_{x_0 + s(x - x_0)}(x - x_0)\,ds$.
  Along $\omega_t = \omega + t(\omega_{x_0} - \omega)$, which is invertible near
  $\{x_0\}\times\mathbb{R}$, Moser's field solves $\omega_t(X_t) = \alpha$; a cut-off
  makes the suspended field $(x,t) \mapsto (X_t x, 1)$ globally Lipschitz.  Its
  flow is $C^1$ with the variational equation
  $\partial_t D\psi_t = DX(\psi_t)\circ D\psi_t$, which Mathlib lacks and is
  proved here from Picard--Lindel\"of and Gr\"onwall.  Differentiating
  $\omega_t(X_t) = \alpha$ in space, the derivative of
  $\omega_t(\psi_t x)(D\psi_t a, D\psi_t b)$ cancels by closedness, alternation and
  $d\alpha = \omega - \omega_{x_0}$, so the time-one map pulls $\omega_{x_0}$ back to
  $\omega$, and the inverse function theorem makes it a chart.
\end{theorem}

\begin{definition}[Periodic orbits and nondegeneracy, \S6.1]
  \label{def:6.1}

  A $1$-periodic orbit through $p$ is exactly a fixed point of the time-$1$ map,
  proved in both directions from ODE uniqueness; nondegeneracy is that $1$ is not
  an eigenvalue of the linearised return map.

  Proposition 6.1.5 --- that a sufficiently Lipschitz field has only constant
  $1$-periodic orbits --- is proved twice.  With the book's sharp constant
  $2\pi$ it is Yorke's theorem, and the proof is the book's: Wirtinger's
  inequality $\int_0^1 |y|^2 \le \tfrac{1}{4\pi^2} \int_0^1 |y'|^2$ for a

  product: for a general Banach norm the sharp bound is $6 < 2\pi$
  (Busenberg--Fisher--Martelli), so the statement carries that hypothesis.  For
  every norm it is proved with the elementary constant $4$: the mean of $\dot x$
  over a period vanishes, so $\dot x(t)$ is an average of differences of $X$
  along the orbit, bounded by $K$ times the mean distance on the circle, which is
  $\tfrac14$.
\end{definition}

\begin{theorem}[The Cauchy--Pompeiu formula and the Cauchy transform]
  \label{thm:cauchy-pompeiu}
  \lean{MorseFloer.CauchyPompeiu.dbar, MorseFloer.CauchyPompeiu.dz,
        MorseFloer.CauchyPompeiu.integral_dbar_div_eq,
        MorseFloer.CauchyPompeiu.cauchyTransform,
        MorseFloer.CauchyPompeiu.cauchyTransform_dbar,
        MorseFloer.CauchyPompeiu.dbar_cauchyTransform}
  \leanok
  For $w$ of class $C^1$ with compact support,
  \[\int_{\mathbb{C}} \frac{(\bar\partial w)(z-\xi)}{\xi}\,d\xi = 2\pi\, w(z),\]
  proved in polar coordinates from the fundamental theorem of calculus: the
  radial integration reconstructs $w$ because the angular average of
  $\xi \mapsto (\bar\partial w)(z - \xi)$ integrates to the value at the centre.
  Consequently the \emph{Cauchy transform}
  $T f (z) = (2\pi)^{-1}\int f(z-\xi)/\xi \,d\xi$ is a two-sided solution
  operator for $\bar\partial v = f$: both $T(\bar\partial w) = w$ and
  $\bar\partial(T f) = f$ hold, the second by differentiating under the integral
  sign, which the local integrability of $1/\xi$ permits.
\end{theorem}

\begin{theorem}[The Beurling transform and the Calder\'on--Zygmund estimate]
  \label{thm:beurling}
  \lean{MorseFloer.CauchyPompeiu.integrableOn_rpow_neg_ball,
        MorseFloer.CauchyPompeiu.beurling,
        MorseFloer.CauchyPompeiu.setIntegral_beurling_kernel_annulus,
        MorseFloer.CauchyPompeiu.beurlingWith_eq,
        MorseFloer.CauchyPompeiu.norm_beurling_le,
        MorseFloer.CauchyPompeiu.norm_beurling_sub_le_holder,
        MorseFloer.CauchyPompeiu.dz_cauchyTransform_eq_beurling}
  \leanok
  \uses{thm:cauchy-pompeiu}
  The \emph{other} derivative of the Cauchy transform is the singular integral
  $B f(z) = \mathrm{p.v.}\int f(z-\xi)\,\xi^{-2}\,d\xi$, and
  $\partial (Tf)/\partial z = -\pi^{-1} B f$.  The principal value converges on
  compactly supported $C^{0,\alpha}$ data --- the Riesz kernel
  $\lVert\xi\rVert^{-a}$ is integrable on a disc exactly for $a < 2$ --- and the
  truncation radius is immaterial because the kernel has zero mean on every
  annulus centred at its pole, by the quarter turn $\xi \mapsto i\xi$, which
  preserves Lebesgue measure and negates the kernel.  The Calder\'on--Zygmund
  estimate holds at the \emph{sharp} exponent: $B$ maps compactly supported
  $C^{0,\alpha}$ data to $C^{0,\alpha}$ for $0 < \alpha < 1$.  Both transforms
  are split at the radius $2\lVert z_1 - z_2\rVert$; the near parts are
  $O(d^\alpha)$; in the far parts, subtracting the constant $f(z_1)$ costs
  nothing because the difference of the two truncated kernels integrates to zero
  over the plane --- the reflection through the midpoint of the two poles
  exchanges them, so no improper limit is needed --- and what remains is bounded
  near the poles by the area of a disc and far from them by the H\"ormander
  condition.  The identification of $\partial (Tf)/\partial z$ is proved in polar
  coordinates, the radial part telescoping to zero and the angular part
  integrating by parts into $B$.
\end{theorem}

\begin{proposition}[The Schauder estimate on the H\"older scale]
  \label{prop:schauder}
  \lean{MorseFloer.CauchyHolder.dbar_cauchyTransform_holder,
        MorseFloer.CauchyHolder.dz_cauchyTransform_holder,
        MorseFloer.CauchyHolder.holder_fderiv_cauchyTransform,
        MorseFloer.CauchyHolder.IsHolderC,
        MorseFloer.CauchyHolder.isHolderC_cauchyTransform,
        MorseFloer.CauchyHolder.contDiff_cauchyTransform_of_isHolderC}
  \leanok
  \uses{thm:beurling}
  The solution operator gains a full derivative without losing the H\"older
  exponent: $T$ maps $C^{k,\alpha}$ to $C^{k+1,\alpha}$, for every $k$ and every
  $0 < \alpha < 1$.  Both identities of \ref{thm:beurling} are first carried from
  $C^1$ data to merely H\"older data by mollification --- mollification preserves
  the H\"older constant and converges uniformly at the rate $\varepsilon^\alpha$,
  and a uniform limit of derivatives is the derivative of the limit.  The scale
  itself is defined recursively through directional derivatives, and the
  induction on $k$ is short because the derivative of $Tf$ in a direction $v$ is
  the fixed combination $\tfrac{v}{2} T(\partial f/\partial z) + \tfrac{\bar
  v}{2} f$, in which $\partial f/\partial z$ sits one level lower.  The $C^k$
  scale alone would not do: the Cauchy transform of a $C^k$ function is only
  $C^k$.
\end{proposition}

\begin{theorem}[The elliptic bootstrap for $\bar\partial$]
  \label{thm:dbar-bootstrap}
  \lean{MorseFloer.CauchyHolder.exists_cutoff_isHolderC,
        MorseFloer.CauchyHolder.contDiff_infty_of_dbar_contDiff,
        MorseFloer.CauchyHolder.exists_isHolderC_comp,
        MorseFloer.CauchyHolder.contDiff_infty_of_dbar_comp,
        MorseFloer.CauchyHolder.exists_isHolderC_comp₂,

  \leanok
  \uses{prop:schauder}
  A $C^1$ function whose $\bar\partial$ is smooth is smooth; more generally a
  $C^1$ solution of the semilinear system $\bar\partial u_i = G_i(z, u_1, \dots,
  u_n)$ with $G$ smooth is smooth.  Weyl's lemma (\ref{lem:12.1.1w}) is not
  needed: a compactly supported $C^1$ function \emph{equals} the Cauchy transform
  of its own $\bar\partial$, so after a cut-off the regularity of the function is
  exactly that of that transform.  The cut-off costs a commutator term which
  involves the function only where the previous cut-off was already $1$, so the
  induction runs on shrinking discs.  The nonlinearity is absorbed by a chain
  rule on the H\"older scale, in one, two and $n$ inner functions; the point
  dependence of $G$ is handled by feeding the point in as one more inner
  function, cut off like the solution; and the vector target needs no
  vector-valued Cauchy transform, since the components of the system are coupled
  only through the right-hand side and climb the scale together.
\end{theorem}

\begin{proposition}[Proposition 6.5.7 and Corollary 6.5.11]
  \label{prop:6.5.7}

  \ref{def:6.4} and $E(u) = \mathcal{A}_H(x) - \mathcal{A}_H(y)$).  Both are

  sequences $s_k \to \pm\infty$ along which
  $\lVert \partial_t u - X_t(u)\rVert_{L^2(S^1)} \to 0$.  The loops

  converge, and the limit solves the integral equation by dominated convergence,

  --- and, being monotone, it then converges along the whole half-line.

\begin{lemma}[A mean value inequality for almost holomorphic functions]
  \label{lem:mean-value}
  \lean{MorseFloer.MeanValue.cut, MorseFloer.MeanValue.exists_norm_dbar_cut_le,
        MorseFloer.MeanValue.norm_le_of_dbar_le}
  \leanok

  $P$ on the plane with $\lvert\bar\partial P\rvert \le a$ on the disc of radius
  $r$ about $z_1$,
  \[\lvert P(z_1)\rvert \le \frac{K_0}{r^2}\int_{\lvert\xi\rvert<r}
    \lvert P(z_1-\xi)\rvert\,d\xi + K_2\,a\,r.\]
  This is the Cauchy--Pompeiu formula (\ref{thm:cauchy-pompeiu}) applied to
  $\chi P$ for a cut-off $\chi$ at scale $r$: $P(z_1) = T(\bar\partial(\chi P))(z_1)$
  and $\bar\partial(\chi P) = \chi\,\bar\partial P + P\,\bar\partial\chi$.  The first
  term is integrated against the Riesz kernel $1/\lvert\xi\rvert$, whose mass on
  the disc is proportional to $r$; the second lives on an annulus, where the
  kernel is at most $2/r$ and $\lvert\bar\partial\chi\rvert$ is $O(1/r)$.
\end{lemma}

  \leanok
  \uses{prop:6.5.3, lem:mean-value}
  The gradient of the solutions of energy at most $C$ is bounded by a constant
  depending only on $C$.  \emph{Proved}, and without the bubbling analysis of the
  book, which needs $C^1_{loc}$ compactness of rescaled solutions and the area
  computations of Lemmas 6.6.4 and 6.6.5.  A solution being smooth
  (\ref{prop:6.5.3}), $p = \partial_s u$ solves the linearised equation
  $\partial_s p + J_0\,\partial_t p + (\mathrm{Hess}\,H_t)(u)\,p = 0$, whose
  zeroth-order term is bounded on the torus; read in $\mathbb{C}^n$, $p$ is
  almost holomorphic, $\lvert\bar\partial p\rvert \le 2L\lvert p\rvert$.  Hofer's
  lemma (Lemma 6.6.3) gives, near any point, a disc of radius
  $\varepsilon \le \tfrac12$ on which $\lvert p\rvert$ is at most twice its value
  $N$ at the centre, with $\varepsilon N$ at least half the value at the given
  point.  On a concentric disc of radius $r \le \varepsilon$ the mean value
  inequality (\ref{lem:mean-value}) and
  $\lvert p\rvert \le \tfrac1{2r} + \tfrac r2\lvert p\rvert^2$ give
  $N \le K_1/r + 4K_2LNr$, the $L^2$ norm on the disc being at most the energy by
  Fubini, since a disc of radius at most $\tfrac12$ lies in one period of the
  cylinder.  For $r$ below a fixed threshold the last term is absorbed, which
  bounds $\varepsilon N$ or $N$ itself.  The bound on $\partial_t u$ follows from
  the Floer equation.  This is the book's rescaling argument with the limit
  taken out: on the torus with the standard $J_0$ the bubble would be an entire
  function with bounded derivative and finite energy.
\end{proposition}

\begin{theorem}[Theorem 6.5.6 and Proposition 6.5.15, convergence at the ends]
  \label{thm:6.5.6}
  \lean{MorseFloer.LatticePath.tendsto_of_tendsto_zero,
        MorseFloer.ApproxOrbit.tendsto_window,
        MorseFloer.Chapter6.exists_norm_dS_le,

  $s \to \pm\infty$ to periodic orbits, and $\partial_s u \to 0$ uniformly in

  decay of $\partial_s u$ is the mean value inequality (\ref{lem:mean-value})
  once more: the $L^1$ norm of $\partial_s u$ on a disc of radius at most
  $\tfrac12$ is controlled by the energy of the window $[s-\tfrac12, s+\tfrac12]$,
  which tends to $0$.  So the loops $u(s,\cdot)$ solve Hamilton's equation up to
  an error tending to $0$ uniformly, and their starting points $q(s) = u(s,0)$
  form a continuous path along which $\psi_1 q - q \to 0$, by Gr\"onwall against

  fixed point, and the fixed points are uniformly separated, both by translating
  into the unit cube; so the path eventually lies in a union of disjoint balls
  about fixed points, hence in one of them, and converges.  Gr\"onwall once more
  makes the loops converge to the orbit through the limit.
\end{theorem}

\begin{theorem}[Theorem 6.5.4, compactness of the space of solutions]
  \label{thm:6.5.4}
  \lean{MorseFloer.LipschitzLimit.exists_subseq_tendstoUniformlyOn,
        MorseFloer.DbarLimit.tendsto_cauchyTransform,
        MorseFloer.DbarLimit.contDiff_one_of_tendsto_dbar,
        MorseFloer.Chapter6.isSmoothLoopVariation_of_contDiff,
        MorseFloer.Chapter6.IsFloerSolution.sub_lattice,
        MorseFloer.Chapter6.lipschitz_of_gradient_bound,
        MorseFloer.Chapter6.compactness_of_energy_bounded}
  \leanok
  \uses{prop:6.6.2, prop:6.5.3, thm:cauchy-pompeiu, prop:schauder}
  A sequence of solutions of uniformly bounded energy has a subsequence which,
  after translation by lattice vectors, converges uniformly on every compact
  subset of the cylinder to a solution.  \emph{Proved}, and without the uniform
  elliptic estimates of the book.  The gradient bound (\ref{prop:6.6.2}) makes
  the translated solutions equi-Lipschitz, and Arzel\`a--Ascoli --- done by
  hand: restricted to the rational points the sequence lives in a compact,
  first-countable product of closed balls, which replaces the diagonal
  argument, and the Lipschitz bound spreads the convergence to compact sets ---
  gives a locally uniformly convergent subsequence with a Lipschitz limit.  Read
  in $\mathbb{C}^n$, each solution satisfies $\bar\partial u_i = G_i(z,u)$, and
  the right-hand sides converge locally uniformly too.  For a cut-off $\chi$,
  $\chi u = T(\chi\,\bar\partial u + u\,\bar\partial\chi)$ by the
  Cauchy--Pompeiu formula (\ref{thm:cauchy-pompeiu}), and both sides pass to the
  limit, the Cauchy transform being continuous under uniform convergence of
  data with supports in a fixed disc; the two data of the limit are Lipschitz
  with compact support, so their transforms are $C^1$ with the prescribed
  $\bar\partial$ (\ref{prop:schauder}).  Hence the limit is $C^1$ and solves the
  limiting equation, the bootstrap (\ref{prop:6.5.3}) makes it smooth, and
  smoothness supplies the partial derivatives the definition of a solution asks
  for.  The convergence of the derivatives is a consequence, not an input.
\end{theorem}

\begin{lemma}[The roots of a polynomial depend continuously on its coefficients]
  \label{lem:roots-continuity}
  \lean{MorseFloer.RootsContinuity.norm_root_le,
        MorseFloer.RootsContinuity.continuous_coeff_prod,
        MorseFloer.RootsContinuity.exists_delta_roots_close,
        MorseFloer.RootsContinuity.exists_delta_countP_roots}
  \leanok
  For a monic polynomial $p$ over $\mathbb{C}$ and every $\varepsilon > 0$ there
  is $\delta > 0$ such that every monic $q$ of the same degree whose coefficients
  are within $\delta$ of those of $p$ has its roots within $\varepsilon$ of those
  of $p$, counted with multiplicity: the two root multisets admit enumerations
  $a, b \colon \{1,\dots,n\} \to \mathbb{C}$ with $\lvert a_i - b_i\rvert <
  \varepsilon$ for all $i$.  Consequently the number of roots of $q$ in a small
  disc about any point equals the multiplicity of that point as a root of $p$.
  This is the first brick for the map $\rho$ of Theorem 7.1.3, whose continuity
  (\S7.3.c) is an argument about the eigenvalues of a varying symplectic
  matrix.  Mathlib has the roots as a multiset but nothing about their
  dependence on the polynomial.  The proof is a compactness argument: if the
  statement failed there would be a sequence $q_k \to p$ with no
  $\varepsilon$-matching of the roots; the roots of the $q_k$ are bounded by
  Cauchy's bound, so an enumeration of them has a convergent subsequence, and
  since the coefficients of $\prod (X - a_i)$ are continuous in $a$, the limiting
  enumeration satisfies $\prod (X - b_i) = p$, so it enumerates the roots of $p$,
  and the matching exists along the subsequence after all.
\end{lemma}

\begin{lemma}[Spectral projectors, continuous in the matrix]
  \label{lem:spectral-projector}
  \lean{MorseFloer.SpectralProjector.continuous_charpoly_coeff,
        MorseFloer.SpectralProjector.fPoly_mul_gPoly,
        MorseFloer.SpectralProjector.isCoprime_fPoly_gPoly,
        MorseFloer.SpectralProjector.continuousAt_coeff_filterProd,
        MorseFloer.SpectralProjector.proj_spec,
        MorseFloer.SpectralProjector.eventually_circle,
        MorseFloer.SpectralProjector.exists_continuous_projector}
  \leanok
  \uses{lem:roots-continuity}
  Let $A_0$ be a complex square matrix and $D$ a disc whose boundary circle
  carries no eigenvalue of $A_0$.  The characteristic polynomial of any $A$
  splits as $f_A g_A$, with $f_A$ collecting the eigenvalues in $D$ and $g_A$ the
  others; the two are coprime, so $\mathbb{C}^n = \ker f_A(A) \oplus \ker g_A(A)$,
  the first summand being the sum of the generalised eigenspaces for the
  eigenvalues in $D$.  There are a neighbourhood $U$ of $A_0$ and a continuous
  $P \colon U \to M_n(\mathbb{C})$ such that $P_A$ is the projector onto
  $\ker f_A(A)$ along $\ker g_A(A)$, commuting with $A$; and the number of
  eigenvalues in $D$, with multiplicity, is constant on $U$.  This is the
  second brick for $\rho$: the continuity argument of \S7.3.c compares the form
  $Q$ on the generalised eigenspaces of nearby matrices, which $P$ transports.
  No contour integral is used.  The projector comes from a B\'ezout relation
  $u_0 f_{A_0} + v_0 g_{A_0} = 1$ at the base point: for $A$ near $A_0$,
  $R_A = u_0 f_A(A) + v_0 g_A(A)$ is close to the identity, hence invertible,
  and $P_A = v_0(A) g_A(A) R_A^{-1}$ has the required properties by pure algebra
  ($f_A g_A$ vanishes at $A$ by Cayley--Hamilton).  Its continuity reduces to
  that of the coefficients of $f_A$ and $g_A$, which follows from the continuity
  of the eigenvalues (\ref{lem:roots-continuity}): the matching of the
  eigenvalues of $A$ with those of $A_0$ preserves the side of the circle, and
  the coefficients of $\prod (X - a_i)$ are uniformly continuous in $a$ on a
  bounded set.  The coefficients of the characteristic polynomial itself are
  continuous in the matrix, by the Leibniz formula.
\end{lemma}

\begin{lemma}[Lemma 7.3.3 and Corollary 7.3.4: the form $H$ on the generalised eigenspaces]
  \label{lem:7.3.3}
  \lean{MorseFloer.SymplecticEigen.HForm, MorseFloer.SymplecticEigen.HForm_cpx,
        MorseFloer.SymplecticEigen.HForm_eq_zero_of_inGen,
        MorseFloer.SymplecticEigen.HForm_eq_zero_of_inGen_self,
        MorseFloer.SymplecticEigen.eq_zero_of_forall_HForm_eq_zero,
        MorseFloer.SymplecticEigen.eq_zero_of_inGen_of_forall_HForm_eq_zero,
        MorseFloer.SymplecticEigen.eq_zero_of_inGen_of_forall_BForm_eq_zero}
  \leanok
  \uses{def:7.1.c}
  On $\mathbb{C}^{2n}$ let $H(X, Y) = \omega(X, \bar Y)$, a sesquilinear form
  whose imaginary part is the real form $B$ of \S7.3.a.  A real symplectic
  matrix, complexified, preserves $H$: it preserves $\omega$ and commutes with
  conjugation.  Lemma 7.3.3: the generalised eigenspaces $E_\mu$ and $E_\nu$ of
  such a matrix are $H$-orthogonal whenever $\mu\bar\nu \ne 1$.  The proof is the
  book's: for $X \in \ker (A-\mu)^k$ and $Y \in \ker (A-\nu)^l$ write
  $AX = \mu X + X'$ and $AY = \nu Y + Y'$ with $X'$, $Y'$ one level down, expand
  $H(X,Y) = H(AX, AY)$, and induct on the levels.  Corollary 7.3.4: $E_\mu$ is
  $H$-isotropic when $\lvert\mu\rvert \ne 1$.  Finally $H$ is nondegenerate on
  $\mathbb{C}^{2n}$, so it is nondegenerate on $E_\nu$ when
  $\lvert\nu\rvert = 1$: a vector of $E_\nu$ orthogonal to $E_\nu$ is orthogonal
  to every other $E_\mu$ by the lemma, hence to everything, the generalised
  eigenspaces spanning $\mathbb{C}^{2n}$; replacing $Y$ by $iY$ gives the same
  for $B$.  This is what makes the signature $m_+(\nu)$ of $Q$ on $E_\nu$
  meaningful, the third brick for $\rho$.
\end{lemma}

Almost every statement of this chapter is an inequality between
$W^{k,p}$-norms on a domain, and Mathlib has no such spaces --- see the note on
Mathlib's actual Sobolev holdings in \ref{thm:16.4}.  What survives is the
classical form of the regularity lemma, the chapter's logical architecture, and

\begin{lemma}[Lemma 12.1.1, distributional form: Weyl's lemma]
  \label{lem:12.1.1w}
  \lean{MorseFloer.Chapter12.IsWeakCauchyRiemann,
        MorseFloer.Chapter12.exists_differentiable_of_isWeakCauchyRiemann,
        MorseFloer.Weyl.exists_differentiableOn_ae_eq,
        MorseFloer.Weyl.integral_moll_smul_eq,
        MorseFloer.Weyl.smooth_eq_smooth,
        MorseFloer.Weyl.ae_tendsto_smooth,
        MorseFloer.Weyl.dbar_smooth_eq_zero}
  \leanok
  \uses{lem:12.1.1}
  A locally integrable weak solution of $\bar\partial u = 0$ agrees almost
  everywhere with a holomorphic function.  Proved by mollification, with no
  elliptic estimate: a radial mollifier $\rho_\varepsilon(w) = \psi(|w|^2)$ is
  built by hand (Mathlib's bump functions do not expose radiality); the
  mollification $\rho_\varepsilon * u$ is smooth and its $\bar\partial$ is the
  weak equation tested against $\rho_\varepsilon(x-\cdot)$, so it is
  holomorphic; a holomorphic function equals its own mollification, by polar
  coordinates, Fubini and Cauchy's formula on each circle, so two
  mollifications agree wherever both are holomorphic (associativity and
  commutativity of convolution); and mollifications converge to $u$ almost
  everywhere by the Lebesgue differentiation theorem.  The local holomorphic
  representatives are glued by uniqueness of continuous representatives on a
  countable cover.
\end{lemma}

\chapter{The second derivative of the Floer operator (Chapter 13)}

Everything phrased in $W^{1,p}(\mathbb{R}\times S^1)$ is unstatable, so the two

\begin{lemma}[Lemma 13.4.1]
  \label{lem:13.4.1}
  \lean{MorseFloer.Chapter13.norm_fderiv_comp_sub_le,
        MorseFloer.Chapter13.norm_fderiv_comp₃_sub_le,
        MorseFloer.Chapter13.norm_fderiv_bilinear_sub_le}

  is that it must not depend on the gluing parameter $\rho$.  A bounded bilinear
  map satisfies hypothesis (3) automatically, which is why Lemma 13.4.2 holds.
\end{lemma}
